#include <Arduino_PortentaMachineControl.h>
#include <Ethernet.h>
#include <SPI.h>

// Calibration and voltage conversion
const int analogPin = 0;
const float slope = 61.066;
const float intercept = -30.826;
const float adcMax = 65535.0;
const float vRef = 2.8;
const float voltageDividerGain = (100.0 + 39.0) / 39.0;

// Ethernet setup
byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };
IPAddress ip(192, 168, 1, 177); // Change to match your network
EthernetServer server(80);

void setup() {
  Serial.begin(115200);
  while (!Serial);

  // Init analog input
  MachineControl_AnalogIn.begin(SensorType::V_0_10);

  // Init Ethernet
  Ethernet.begin(mac, ip);
  server.begin();
  Serial.print("Web server running at http://");
  Serial.println(Ethernet.localIP());
}

void loop() {
  EthernetClient client = server.available();

  if (client) {
    // Wait for data from client
    while (client.connected()) {
      if (client.available()) {
        // Read request and discard
        String req = client.readStringUntil('\r');
        client.read(); // remove '\n'

        // Read sensor
        uint16_t raw = MachineControl_AnalogIn.read(analogPin);
        float voltage = (raw / adcMax) * vRef * voltageDividerGain;
        float load = slope * voltage + intercept;

        // Build response
        String html = "<!DOCTYPE html><html><head><meta http-equiv='refresh' content='1'/>";
        html += "<title>Strain Gauge Monitor</title></head><body>";
        html += "<h2>Portenta Machine Control</h2>";
        html += "<p><strong>Voltage:</strong> " + String(voltage, 3) + " V</p>";
        html += "<p><strong>Load:</strong> " + String(load, 2) + " lb</p>";
        html += "</body></html>";

        // Send HTTP response
        client.println("HTTP/1.1 200 OK");
        client.println("Content-Type: text/html");
        client.println("Connection: close");
        client.println();
        client.println(html);
        break;
      }
    }
    delay(1); // Short delay before disconnect
    client.stop();
  }
}
